USAGE EXAMPLES
================================================================================

# Initialize
from acd.application.release import ReleaseManager
manager = ReleaseManager(session, database_path)

# Check for updates
status = manager.check_for_updates(include_beta=False)
if status['update_available']:
    print(f"Update available: {status['latest_version']}")

# Perform update
result = manager.perform_update("0.4.0", "0.5.0")

# Use feature flags
manager.enable_feature("experimental_feature")
if manager.is_feature_enabled("experimental_feature"):
    # Use feature...

# Search help
topics = manager.search_help("installation")

# Load plugins
result = manager.load_plugins(["./plugins"])
